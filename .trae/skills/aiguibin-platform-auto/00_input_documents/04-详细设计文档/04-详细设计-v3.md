```java
@RestController
@RequestMapping("/api/auth")
@Slf4j
public class AuthController {
    
    @Autowired
    private AuthService authService;
    
    /**
     * 第一步登录 - 账号密码验证
     */
    @PostMapping("/login")

     * 第二步登录 - 选择机构
```

#### 1.1.2 认证服务实现
```java
/**
 * 认证服务实现
 */
@Service
@Slf4j
public class AuthServiceImpl implements AuthService {
    
    @Autowired
    private SysUserService userService;
    
    @Autowired
    private SysUserOrgService userOrgService;
    
    @Autowired
    private SysOrgService orgService;
    
    @Autowired
    private SysMenuService menuService;
    
    @Autowired
    private JwtTokenProvider tokenProvider;
    
    @Autowired
    private PasswordEncoder passwordEncoder;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Value("${security.max-login-fail:5}")
    private int maxLoginFail;
    
    /**
     * 第一步登录
     */
    @Override
    public LoginStep1Response loginStep1(LoginRequest request, String ip) {
        String username = request.getUsername();
        
        // 1. 查询用户
        SysUser user = userService.getByUsername(username);
        if (user == null) {
            log.warn("登录失败：用户不存在, username={}", username);
            throw new BusinessException("用户名或密码错误");
        }
        
        // 2. 检查账号是否锁定
        if (user.getIsLocked() != null && user.getIsLocked() == 1) {
            log.warn("登录失败：账号已锁定, username={}", username);
            throw new BusinessException("账号已锁定，请联系管理员");
        }
        
        // 3. 验证密码
        if (!passwordEncoder.matches(request.getPassword(), user.getPassword())) {
            // 记录登录失败次数
            recordLoginFail(user);
            log.warn("登录失败：密码错误, username={}", username);
            throw new BusinessException("用户名或密码错误");
        }
        
        // 4. 获取用户所属机构列表
        List<UserOrgVO> orgList = userOrgService.getOrgListByUserId(user.getId());
        if (CollectionUtils.isEmpty(orgList)) {
            log.warn("登录失败：用户未分配机构, username={}", username);
            throw new BusinessException("用户未分配机构，请联系管理员");
        }
        
        // 5. 生成临时Token
        String tempToken = tokenProvider.generateTempToken(user.getId());
        
        // 6. 缓存用户信息到Redis
        String cacheKey = "login:temp:" + tempToken;
        redisTemplate.opsForValue().set(cacheKey, user, 5, TimeUnit.MINUTES);
        
        // 7. 更新登录信息
        userService.updateLoginInfo(user.getId(), ip);
        
        // 8. 清除登录失败记录
        clearLoginFail(user.getId());
        
        log.info("第一步登录成功：username={}, orgCount={}", username, orgList.size());
        
        return LoginStep1Response.builder()
            .tempToken(tempToken)
            .orgList(orgList)
            .build();
    }
    
    /**
     * 第二步登录 - 选择机构
     */
    @Override
    public LoginResponse loginStep2(SelectOrgRequest request) {
        String tempToken = request.getTempToken();
        String orgCode = request.getOrgCode();
        
        // 1. 从Redis获取临时Token缓存的用户信息
        String cacheKey = "login:temp:" + tempToken;
        SysUser user = (SysUser) redisTemplate.opsForValue().get(cacheKey);
        if (user == null) {
            throw new BusinessException("登录已过期，请重新登录");
        }
        
        // 2. 验证用户是否有该机构权限
        boolean hasOrg = userOrgService.checkUserOrg(user.getId(), orgCode);
        if (!hasOrg) {
            log.warn("登录失败：用户无该机构权限, userId={}, orgCode={}", user.getId(), orgCode);
            throw new BusinessException("无该机构权限");
        }
        
        // 3. 获取机构信息
        SysOrg org = orgService.getByCode(orgCode);
        
        // 4. 获取用户角色和权限
        List<String> roles = userService.getUserRoles(user.getId());
        List<String> permissions = userService.getUserPermissions(user.getId());
        
        // 5. 构建登录用户信息
        LoginUser loginUser = LoginUser.builder()
            .userId(user.getId())
            .userNo(user.getUserNo())
            .username(user.getUsername())
            .realName(user.getRealName())
            .orgCode(orgCode)
            .orgName(org != null ? org.getOrgName() : null)
            .deptCode(user.getDeptCode())
            .roles(roles)
            .permissions(permissions)
            .dataScope(userService.getDataScope(user.getId()))
            .build();
        
        // 6. 生成Token
        String accessToken = tokenProvider.generateAccessToken(loginUser);
        String refreshToken = tokenProvider.generateRefreshToken(user.getId());
        String csrfToken = tokenProvider.generateCsrfToken();
        
        // 7. 缓存登录状态
        String loginKey = "login:user:" + user.getId();
        redisTemplate.opsForValue().set(loginKey, loginUser, 30, TimeUnit.MINUTES);
        
        // 8. 删除临时Token
        redisTemplate.delete(cacheKey);
        
        log.info("登录成功：username={}, orgCode={}", user.getUsername(), orgCode);
        
        return LoginResponse.builder()
            .accessToken(accessToken)
            .refreshToken(refreshToken)
            .csrfToken(csrfToken)
            .expiresIn(1800)
            .userInfo(loginUser)
            .build();
    }
    
    /**
     * 记录登录失败次数
     */
    private void recordLoginFail(SysUser user) {
        String failKey = "login:fail:" + user.getId();
        Long failCount = redisTemplate.opsForValue().increment(failKey);
        redisTemplate.expire(failKey, 30, TimeUnit.MINUTES);
        
        if (failCount != null && failCount >= maxLoginFail) {
            userService.lockUser(user.getId(), "密码错误次数过多，系统自动锁定");
            log.warn("用户账号已锁定：username={}, failCount={}", user.getUsername(), failCount);
        }
    }
    
    /**
     * 清除登录失败记录
     */
    private void clearLoginFail(Long userId) {
        String failKey = "login:fail:" + userId;
        redisTemplate.delete(failKey);
    }
}
```

### 1.2 JWT Token实现

```java
/**
 * JWT Token提供者
 */
@Component
@Slf4j
public class JwtTokenProvider {
    
    @Value("${jwt.secret:your-secret-key-here}")
    private String jwtSecret;
    
    @Value("${jwt.access-token-expiration:1800000}")  // 30分钟
    private long accessTokenExpiration;
    
    @Value("${jwt.refresh-token-expiration:604800000}")  // 7天
    private long refreshTokenExpiration;
    
    private SecretKey key;
    
    @PostConstruct
    public void init() {
        this.key = Keys.hmacShaKeyFor(jwtSecret.getBytes(StandardCharsets.UTF_8));
    }
    
    /**
     * 生成临时Token（第一步登录）
     */
    public String generateTempToken(Long userId) {
        Date now = new Date();
        Date expiry = new Date(now.getTime() + 300000); // 5分钟
        
        return Jwts.builder()
            .setSubject(String.valueOf(userId))
            .setIssuedAt(now)
            .setExpiration(expiry)
            .claim("type", "TEMP")
            .signWith(key, SignatureAlgorithm.HS256)
            .compact();
    }
    
    /**
     * 生成访问Token
     */
    public String generateAccessToken(LoginUser user) {
        Date now = new Date();
        Date expiry = new Date(now.getTime() + accessTokenExpiration);
        
        return Jwts.builder()
            .setSubject(String.valueOf(user.getUserId()))
            .setIssuedAt(now)
            .setExpiration(expiry)
            .claim("type", "ACCESS")
            .claim("userNo", user.getUserNo())
            .claim("username", user.getUsername())
            .claim("realName", user.getRealName())
            .claim("orgCode", user.getOrgCode())
            .claim("deptCode", user.getDeptCode())
            .claim("roles", user.getRoles())
            .claim("permissions", user.getPermissions())
            .claim("dataScope", user.getDataScope())
            .signWith(key, SignatureAlgorithm.HS256)
            .compact();
    }
    
    /**
     * 生成刷新Token
     */
    public String generateRefreshToken(Long userId) {
        Date now = new Date();
        Date expiry = new Date(now.getTime() + refreshTokenExpiration);
        
        return Jwts.builder()
            .setSubject(String.valueOf(userId))
            .setIssuedAt(now)
            .setExpiration(expiry)
            .claim("type", "REFRESH")
            .signWith(key, SignatureAlgorithm.HS256)
            .compact();
    }
    
    /**
     * 生成CSRF Token
     */
    public String generateCsrfToken() {
        return UUID.randomUUID().toString().replace("-", "");
    }
    
    /**
     * 解析Token
     */
    public Claims parseToken(String token) {
        return Jwts.parserBuilder()
            .setSigningKey(key)
            .build()
            .parseClaimsJws(token)
            .getBody();
    }
    
    /**
     * 验证Token是否有效
     */
    public boolean validateToken(String token) {
        try {
            Jwts.parserBuilder().setSigningKey(key).build().parseClaimsJws(token);
            return true;
        } catch (Exception e) {
            return false;
        }
    }
    
    /**
     * 从Token获取用户ID
     */
    public Long getUserIdFromToken(String token) {
        Claims claims = parseToken(token);
        return Long.valueOf(claims.getSubject());
    }
}
```

### 1.3 Token验证过滤器

```java
/**
 * JWT认证过滤器
 */
@Component
@Slf4j
public class JwtAuthenticationFilter extends OncePerRequestFilter {
    
    @Autowired
    private JwtTokenProvider tokenProvider;
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    @Value("${security.csrf-enabled:true}")
    private boolean csrfEnabled;
    
    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                     HttpServletResponse response,
                                     FilterChain filterChain) throws ServletException, IOException {
        
        String requestUri = request.getRequestURI();
        
        // 放行登录相关接口
        if (isPublicUrl(requestUri)) {
            filterChain.doFilter(request, response);
            return;
        }
        
        try {
            // 1. 获取Token
            String token = getTokenFromRequest(request);
            if (StringUtils.isBlank(token)) {
                throw new AuthenticationException("未提供认证Token");
            }
            
            // 2. 验证Token有效性
            if (!tokenProvider.validateToken(token)) {
                throw new AuthenticationException("Token无效或已过期");
            }
            
            // 3. 解析Token
            Claims claims = tokenProvider.parseToken(token);
            
            // 4. 验证CSRF Token（写操作）
            if (csrfEnabled && isWriteOperation(request)) {
                String csrfToken = request.getHeader("X-CSRF-Token");
                String cachedCsrf = (String) claims.get("csrfToken");
                if (!StringUtils.equals(csrfToken, cachedCsrf)) {
                    throw new AuthenticationException("CSRF Token验证失败");
                }
            }
            
            // 5. 构建登录用户信息
            Long userId = Long.valueOf(claims.getSubject());
            LoginUser loginUser = buildLoginUser(claims, userId);
            
            // 6. 设置安全上下文
            UsernamePasswordAuthenticationToken authentication =
                new UsernamePasswordAuthenticationToken(loginUser, null, loginUser.getAuthorities());
            authentication.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
            SecurityContextHolder.getContext().setAuthentication(authentication);
            
        } catch (Exception e) {
            log.error("认证失败: {}", e.getMessage());
            SecurityContextHolder.clearContext();
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            response.setContentType("application/json;charset=UTF-8");
            response.getWriter().write("{\"code\":401,\"message\":\"" + e.getMessage() + "\"}");
            return;
        }
        
        filterChain.doFilter(request, response);
    }
    
    private String getTokenFromRequest(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (StringUtils.isNotBlank(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7);
        }
        return null;
    }
    
    private boolean isPublicUrl(String uri) {
        return uri.startsWith("/api/auth/") ||
               uri.equals("/") ||
               uri.startsWith("/static/");
    }
    
    private boolean isWriteOperation(HttpServletRequest request) {
        String method = request.getMethod();
        return "POST".equals(method) || "PUT".equals(method) || 
               "DELETE".equals(method) || "PATCH".equals(method);
    }
    
    private LoginUser buildLoginUser(Claims claims, Long userId) {
        @SuppressWarnings("unchecked")
        List<String> roles = (List<String>) claims.get("roles");
        @SuppressWarnings("unchecked")
        List<String> permissions = (List<String>) claims.get("permissions");
        
        return LoginUser.builder()
            .userId(userId)
            .userNo((String) claims.get("userNo"))
            .username((String) claims.get("username"))
            .realName((String) claims.get("realName"))
            .orgCode((String) claims.get("orgCode"))
            .deptCode((String) claims.get("deptCode"))
            .roles(roles != null ? roles : Collections.emptyList())
            .permissions(permissions != null ? permissions : Collections.emptyList())
            .dataScope((Integer) claims.get("dataScope"))
            .build();
    }
}
```

### 1.5 数据权限实现

```java
/**
 * 数据权限处理器
 */
@Component
public class DataPermissionHandler {
    
    /**
     * 构建数据权限SQL条件
     */
    public String buildDataScopeCondition(LoginUser user, String orgColumn, String deptColumn) {
        Integer dataScope = user.getDataScope();
        if (dataScope == null) {
            dataScope = 4; // 默认本人
        }
        
        StringBuilder condition = new StringBuilder();
        
        switch (dataScope) {
            case 1: // 全部数据
                return "";
            case 2: // 本机构及下级
                condition.append(orgColumn).append(" IN (")
                    .append("SELECT org_code FROM sys_org WHERE org_code = '")
                    .append(user.getOrgCode()).append("' OR parent_code LIKE '")
                    .append(user.getOrgCode()).append("%')");
                break;
            case 3: // 本部门
                condition.append(deptColumn).append(" = '").append(user.getDeptCode()).append("'");
                break;
            case 4: // 本人
            default:
                condition.append("create_by = ").append(user.getUserId());
                break;
        }
        
        return condition.toString();
    }
}

/**
 * 数据权限注解
 */
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface DataScope {
    String orgColumn() default "org_code";
    String deptColumn() default "dept_code";
}

/**
 * 数据权限切面
 */
@Aspect
@Component
@Slf4j
public class DataScopeAspect {
    
    @Autowired
    private DataPermissionHandler dataPermissionHandler;
    
    @Before("@annotation(dataScope)")
    public void before(JoinPoint point, DataScope dataScope) {
        // 获取当前用户
        LoginUser user = getCurrentUser();
        if (user == null) {
            return;
        }
        
        // 构建数据权限条件
        String condition = dataPermissionHandler.buildDataScopeCondition(
            user, dataScope.orgColumn(), dataScope.deptColumn());
        
        // 将条件存入ThreadLocal
        DataScopeContext.setCondition(condition);
    }
    
    @After("@annotation(com.aiguibin.platform.core.permission.annotation.DataScope)")
    public void after() {
        DataScopeContext.clear();
    }
    
    private LoginUser getCurrentUser() {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication != null && authentication.getPrincipal() instanceof LoginUser) {
            return (LoginUser) authentication.getPrincipal();
        }
        return null;
    }
}
```

```java
/**
 * 目录Sheet解析器
 * 起始行: 第15行 (索引14)
 * 关键列: 12-是否贯标
 */
@Component
public class MenuSheetParser {
    
    private static final int START_ROW = 14;  // 第15行
    
    /**
     * 列索引定义
     */
    private static final int COL_SEQ = 0;           // 序号
    private static final int COL_DATABASE = 1;      // 所属数据库
    private static final int COL_MODULE = 2;        // 所属模块
    private static final int COL_OWNER = 3;         // 责任人
    private static final int COL_TABLE_EN = 4;      // 表名（英文）
    private static final int COL_TABLE_ZH = 5;      // 表名（中文）
    private static final int COL_SHARD = 6;         // 是否分片
    private static final int COL_SHARD_KEY = 7;     // 分片键
    private static final int COL_INIT = 8;          // 是否初始化表
    private static final int COL_INIT_COUNT = 9;    // 初始化数量
    private static final int COL_MIGRATE = 10;      // 是否迁移表
    private static final int COL_SYNC = 11;         // 是否同步查询库
    private static final int COL_STANDARD = 12;     // 是否贯标 ★关键字段
    private static final int COL_CLEAN = 13;        // 是否数据清理
    private static final int COL_CLEAN_POLICY = 14; // 清理策略
    private static final int COL_ARCHIVE = 15;      // 归档策略
    private static final int COL_REMARK = 16;       // 备注说明
    
    public List<MenuItem> parse(Sheet sheet) {
        List<MenuItem> list = new ArrayList<>();
        
        for (int i = START_ROW; i <= sheet.getLastRowNum(); i++) {
            Row row = sheet.getRow(i);
            if (row == null) continue;
            
            String tableName = getCellValue(row.getCell(COL_TABLE_EN));
            if (StringUtils.isBlank(tableName)) continue;
            
            MenuItem item = new MenuItem();
            item.setSeq(getCellValue(row.getCell(COL_SEQ)));
            item.setDatabaseName(getCellValue(row.getCell(COL_DATABASE)));
            item.setModule(getCellValue(row.getCell(COL_MODULE)));
            item.setOwner(getCellValue(row.getCell(COL_OWNER)));
            item.setTableName(tableName.trim().toUpperCase());
            item.setTableComment(getCellValue(row.getCell(COL_TABLE_ZH)));
            item.setIsShard(getCellValue(row.getCell(COL_SHARD)));
            item.setShardKey(getCellValue(row.getCell(COL_SHARD_KEY)));
            item.setIsInit(getCellValue(row.getCell(COL_INIT)));
            item.setInitCount(getCellValue(row.getCell(COL_INIT_COUNT)));
            item.setIsMigrate(getCellValue(row.getCell(COL_MIGRATE)));
            item.setIsSync(getCellValue(row.getCell(COL_SYNC)));
            item.setIsStandard(getCellValue(row.getCell(COL_STANDARD)));  // 是否贯标
            item.setIsClean(getCellValue(row.getCell(COL_CLEAN)));
            item.setCleanPolicy(getCellValue(row.getCell(COL_CLEAN_POLICY)));
            item.setArchivePolicy(getCellValue(row.getCell(COL_ARCHIVE)));
            item.setRemark(getCellValue(row.getCell(COL_REMARK)));
            
            list.add(item);
        }
        
        return list;
    }
    
    private String getCellValue(Cell cell) {
        if (cell == null) return "";
        cell.setCellType(CellType.STRING);
        return StringUtils.trim(cell.getStringCellValue());
    }
}
```

```java
/**
 * 字段Sheet解析器
 * 每个表对应一个Sheet
 * 起始行: 第1行 (索引1，跳过表头)
 */
@Component
public class ColumnSheetParser {
    
    private static final int START_ROW = 1;
    
    /**
     * 列索引定义
     */
    private static final int COL_SEQ = 0;           // 序号
    private static final int COL_TABLE_EN = 1;      // 表英文名
    private static final int COL_TABLE_ZH = 2;      // 表中文名
    private static final int COL_COLUMN_EN = 3;     // 字段英文名
    private static final int COL_COLUMN_TYPE = 4;   // 字段类型
    private static final int COL_COLUMN_ZH = 5;     // 字段中文名
    private static final int COL_NULLABLE = 6;      // 是否为空
    private static final int COL_PK = 7;            // 是否主键
    private static final int COL_DICT = 8;          // 数据字典
    private static final int COL_BUS_PK = 9;        // 是否业务主键
    private static final int COL_BUS_REQUIRED = 10; // 是否业务必输
    private static final int COL_CHANGE_DATE = 11;  // 变更日期
    private static final int COL_REMARK = 12;       // 备注说明
    
    public Map<String, List<ColumnItem>> parseAll(Workbook workbook, List<MenuItem> menuList) {
        Map<String, List<ColumnItem>> result = new HashMap<>();
        
        for (MenuItem menu : menuList) {
            String sheetName = findSheetName(workbook, menu.getTableComment());
            if (sheetName == null) continue;
            
            Sheet sheet = workbook.getSheet(sheetName);
            if (sheet == null) continue;
            
            List<ColumnItem> columns = parseSheet(sheet, menu.getTableName());
            result.put(menu.getTableName().toUpperCase(), columns);
        }
        
        return result;
    }
    
    private List<ColumnItem> parseSheet(Sheet sheet, String defaultTableName) {
        List<ColumnItem> list = new ArrayList<>();
        
        for (int i = START_ROW; i <= sheet.getLastRowNum(); i++) {
            Row row = sheet.getRow(i);
            if (row == null) continue;
            
            String columnName = getCellValue(row.getCell(COL_COLUMN_EN));
            if (StringUtils.isBlank(columnName)) continue;
            
            ColumnItem item = new ColumnItem();
            item.setSeq(getCellValue(row.getCell(COL_SEQ)));
            item.setTableName(StringUtils.defaultIfBlank(
                getCellValue(row.getCell(COL_TABLE_EN)), defaultTableName).toUpperCase());
            item.setTableComment(getCellValue(row.getCell(COL_TABLE_ZH)));
            item.setColumnName(columnName.trim().toUpperCase());
            item.setColumnType(getCellValue(row.getCell(COL_COLUMN_TYPE)));
            item.setColumnComment(getCellValue(row.getCell(COL_COLUMN_ZH)));
            item.setIsNullable(getCellValue(row.getCell(COL_NULLABLE)));
            item.setIsPk(getCellValue(row.getCell(COL_PK)));
            item.setDictCode(getCellValue(row.getCell(COL_DICT)));
            item.setIsBusPk(getCellValue(row.getCell(COL_BUS_PK)));
            item.setIsBusRequired(getCellValue(row.getCell(COL_BUS_REQUIRED)));
            item.setChangeDate(getCellValue(row.getCell(COL_CHANGE_DATE)));
            item.setRemark(getCellValue(row.getCell(COL_REMARK)));
            
            list.add(item);
        }
        
        return list;
    }
    
    /**
     * 根据表中文名查找Sheet名称
     */
    private String findSheetName(Workbook workbook, String tableComment) {
        if (StringUtils.isBlank(tableComment)) return null;
        
        // 精确匹配
        Sheet exactSheet = workbook.getSheet(tableComment);
        if (exactSheet != null) return tableComment;
        
        // 模糊匹配（Sheet名包含表中文名）
        for (int i = 0; i < workbook.getNumberOfSheets(); i++) {
            String sheetName = workbook.getSheetName(i);
            if (sheetName.contains(tableComment) || tableComment.contains(sheetName)) {
                return sheetName;
            }
        }
        
        return null;
    }
}
```

```java
 * 起始行: 第1行
 */
@Component
public class IndexSheetParser {
    
    private static final int START_ROW = 1;
    
    /**
     * 列索引定义
     */
    private static final int COL_TABLE_EN = 0;      // 表名称
    private static final int COL_TABLE_ZH = 1;      // 表名称中文
    private static final int COL_INDEX_NAME = 2;    // 索引名称
    private static final int COL_INDEX_TYPE = 3;    // 索引类型
    private static final int COL_INDEX_COLUMN = 4;  // 索引字段
    private static final int COL_REMARK = 5;        // 索引说明
    
    public Map<String, List<IndexItem>> parse(Sheet sheet) {
        Map<String, List<IndexItem>> result = new HashMap<>();
        
        for (int i = START_ROW; i <= sheet.getLastRowNum(); i++) {
            Row row = sheet.getRow(i);
            if (row == null) continue;
            
            String tableName = getCellValue(row.getCell(COL_TABLE_EN));
            if (StringUtils.isBlank(tableName)) continue;
            
            IndexItem item = new IndexItem();
            item.setTableName(tableName.trim().toUpperCase());
            item.setTableComment(getCellValue(row.getCell(COL_TABLE_ZH)));
            item.setIndexName(getCellValue(row.getCell(COL_INDEX_NAME)));
            item.setIndexType(getCellValue(row.getCell(COL_INDEX_TYPE)));
            item.setIndexColumn(getCellValue(row.getCell(COL_INDEX_COLUMN)));
            item.setRemark(getCellValue(row.getCell(COL_REMARK)));
            
            result.computeIfAbsent(item.getTableName(), k -> new ArrayList<>()).add(item);
        }
        
        return result;
    }
```

### 4.1 多数据库比对服务
```java
/**
 * 数据库比对服务
 * 支持单环境多数据库比对
 */
@Service
@Slf4j
public class DbCompareService {
    
    @Autowired
    private DatasourceGroupService dsGroupService;
    
    @Autowired
    private ExcelParseService excelParseService;
    
    @Autowired
    private SvnService svnService;
    
    @Autowired
    private NotifyService notifyService;
    
    /**
     * 执行环境比对（该环境所有数据库）
     * @param env 环境标识 (DEV/SIT/UAT)
     * @param triggerType 触发类型 (MANUAL/SCHEDULE)
     */
    public EnvCompareResult compareEnv(String env, String triggerType) {
        log.info("开始{}环境比对, 触发类型: {}", env, triggerType);
        
        // 1. SVN更新
        svnService.updateDoc();
        
        // 2. 解析Excel
        DocData docData = excelParseService.parseDoc();
        
        // 3. 获取该环境所有数据源
        List<String> dataSources = dsGroupService.getDataSourcesByEnv(env);
        log.info("{}环境共有{}个数据库需要比对", env, dataSources.size());
        
        // 4. 循环执行各数据库比对
        List<DbCompareResult> results = new ArrayList<>();
        for (String dsKey : dataSources) {
            try {
                DbCompareResult result = compareSingleDb(env, dsKey, docData);
                results.add(result);
                
                // 发送单个数据库完成通知
                notifyService.sendDbCompleteNotify(env, dsKey, result);
            } catch (Exception e) {
                log.error("{}环境{}数据库比对失败", env, dsKey, e);
                results.add(DbCompareResult.fail(dsKey, e.getMessage()));
            }
        }
        
        // 5. 汇总结果
        EnvCompareResult envResult = new EnvCompareResult();
        envResult.setEnv(env);
        envResult.setDbResults(results);
        envResult.setTotalDbCount(dataSources.size());
        envResult.setSuccessCount((int) results.stream().filter(DbCompareResult::isSuccess).count());
        
        log.info("{}环境比对完成, 成功: {}/{}" , env, envResult.getSuccessCount(), envResult.getTotalDbCount());
        
        return envResult;
    }
    
    /**
     * 执行单个数据库比对
     */
    @DS("#dsKey")
    public DbCompareResult compareSingleDb(String env, String dsKey, DocData docData) {
        String dbName = dsGroupService.getDatabaseName(dsKey);
        log.info("开始比对数据库: {} ({})", dsKey, dbName);
        
        long startTime = System.currentTimeMillis();
        
        // 1. 表比对
        TableCompareResult tableResult = compareTables(dbName, docData);
        
        // 2. 字段比对
        ColumnCompareResult columnResult = compareColumns(dbName, docData);
        
        // 3. 索引比对
        IndexCompareResult indexResult = compareIndexes(dbName, docData);
        
        // 4. 贯标检查（只对贯标表）
        StandardCheckResult standardResult = null;
        if (hasStandardTables(docData)) {
            standardResult = checkStandard(dbName, docData);
        }
        
        // 5. 生成结果文件
        String resultPath = generateResultFile(env, dsKey, dbName, 
            tableResult, columnResult, indexResult, standardResult);
        
        // 6. SVN上传
        svnService.uploadResult(env, dsKey, resultPath);
        
        long duration = System.currentTimeMillis() - startTime;
        
        return DbCompareResult.success(dsKey, dbName, duration, 
            tableResult, columnResult, indexResult, standardResult, resultPath);
    }
    
    /**
     * 执行全环境比对
     */
    public AllEnvCompareResult compareAllEnv(String triggerType) {
        log.info("开始全环境比对");
        
        AllEnvCompareResult allResult = new AllEnvCompareResult();
        allResult.setStartTime(LocalDateTime.now());
        
        // 顺序执行: DEV -> SIT -> UAT
        allResult.setDevResult(compareEnv("DEV", triggerType));
        allResult.setSitResult(compareEnv("SIT", triggerType));
        allResult.setUatResult(compareEnv("UAT", triggerType));
        
        allResult.setEndTime(LocalDateTime.now());
        
        // 发送汇总通知
        notifyService.sendAllEnvCompleteNotify(allResult);
        
        return allResult;
    }
    
    /**
     * 判断是否包含贯标表
     */
    private boolean hasStandardTables(DocData docData) {
        return docData.getMenuList().stream()
            .anyMatch(m -> "Y".equalsIgnoreCase(m.getIsStandard()));
    }
}
```

```java
@Service
@Slf4j
public class SvnService {
    
    @Value("${svn.url}")
    private String svnUrl;
    
    @Value("${svn.work-copy-path}")
    private String workCopyPath;
    
    @Autowired
    private SVNClientManager svnClientManager;
    
    /**
     * 更新设计文档
     */
    public void updateDoc() {
        try {
            File wcDir = new File(workCopyPath);
            SVNUpdateClient updateClient = svnClientManager.getUpdateClient();
            long revision = updateClient.doUpdate(wcDir, SVNRevision.HEAD, SVNDepth.INFINITY, false, false);
            log.info("SVN更新完成, 版本: {}", revision);
        } catch (SVNException e) {
            throw new RuntimeException("SVN更新失败", e);
        }
    }
    
    /**
     * 上传结果文件
     */
    public void uploadResult(String env, String dbKey, String localFilePath) {
        try {
            // 提取数据库名
            String dbName = dbKey.substring(dbKey.lastIndexOf("-") + 1);
            
            // 构建目标目录
            String envDir = workCopyPath + "/results/" + env;
            String dbDir = envDir + "/" + dbName;
            File dbDirFile = new File(dbDir);
            if (!dbDirFile.exists()) {
                dbDirFile.mkdirs();
            }
            
            // 复制文件
            File localFile = new File(localFilePath);
            File destFile = new File(dbDir + "/" + localFile.getName());
            FileUtils.copyFile(localFile, destFile);
            
            // SVN添加
            SVNWCClient wcClient = svnClientManager.getWCClient();
            wcClient.doAdd(destFile, false, false, false, SVNDepth.INFINITY, false, false);
            
            // SVN提交
            SVNCommitClient commitClient = svnClientManager.getCommitClient();
            SVNCommitInfo commitInfo = commitClient.doCommit(
                new File[]{destFile}, 
                false, 
                "[" + env + "][" + dbName + "]数据库比对结果 - " + LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss")),
                null, 
                null
            );
            
            log.info("SVN提交完成, 版本: {}", commitInfo.getNewRevision());
        } catch (Exception e) {
            throw new RuntimeException("SVN上传失败", e);
        }
    }
}
```
